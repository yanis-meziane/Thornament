-- 1. CRÉATION DES TABLES DE BASE (Sans dépendances)
CREATE TABLE public.users (
    id SERIAL PRIMARY KEY,
    mail character varying(255) NOT NULL UNIQUE,
    password character varying(255) NOT NULL,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    created_by integer REFERENCES public.users(id)
);

CREATE TABLE public.tournament (
    id SERIAL PRIMARY KEY,
    name character varying(255) NOT NULL,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    created_by integer NOT NULL REFERENCES public.users(id)
);

CREATE TABLE public.step_settings (
    id SERIAL PRIMARY KEY,
    victory_condition character varying(255),
    number_players integer,
    number_winners integer,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    created_by integer NOT NULL REFERENCES public.users(id)
);

-- 2. STRUCTURE DES ÉTAPES
CREATE TABLE public.step (
    id SERIAL PRIMARY KEY,
    name character varying(255),
    description text,
    tournament_id integer NOT NULL REFERENCES public.tournament(id) ON DELETE CASCADE,
    step_position integer,
    step_component_type character varying(50) NOT NULL,
    step_setting_id integer REFERENCES public.step_settings(id),
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    created_by integer NOT NULL REFERENCES public.users(id),
    CONSTRAINT step_step_component_type_check CHECK (step_component_type IN ('tree', 'br', 'league'))
);

-- 3. JOUEURS ET COMPÉTITION
CREATE TABLE public.players (
    id SERIAL PRIMARY KEY,
    name character varying(255) NOT NULL,
    nationality character varying(100),
    team character varying(100),
    description text,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    created_by integer NOT NULL REFERENCES public.users(id)
);

CREATE TABLE public.takes_place_in_br (
    player_id integer NOT NULL REFERENCES public.players(id) ON DELETE CASCADE,
    br_id integer NOT NULL REFERENCES public.step(id) ON DELETE CASCADE,
    value integer,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    created_by integer NOT NULL REFERENCES public.users(id),
    PRIMARY KEY (player_id, br_id)
);

CREATE TABLE public.versus (
    id SERIAL PRIMARY KEY,
    player1_id integer REFERENCES public.players(id) ON DELETE SET NULL,
    player2_id integer REFERENCES public.players(id) ON DELETE SET NULL,
    player1_value integer,
    player2_value integer,
    step_id integer NOT NULL REFERENCES public.step(id) ON DELETE CASCADE,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    created_by integer NOT NULL REFERENCES public.users(id)
);

CREATE TABLE public.versus_tree (
    id SERIAL PRIMARY KEY,
    player1_id integer REFERENCES public.players(id) ON DELETE SET NULL,
    player2_id integer REFERENCES public.players(id) ON DELETE SET NULL,
    player1_value integer,
    player2_value integer,
    tree_position integer,
    tree_id integer NOT NULL REFERENCES public.step(id) ON DELETE CASCADE,
    created_at timestamp without time zone DEFAULT CURRENT_TIMESTAMP,
    created_by integer NOT NULL REFERENCES public.users(id)
);

-- 4. INDEX (Pour les performances)
CREATE INDEX idx_step_tournament_id ON public.step(tournament_id);
CREATE INDEX idx_takes_place_in_br_br_id ON public.takes_place_in_br(br_id);
CREATE INDEX idx_versus_step_id ON public.versus(step_id);
CREATE INDEX idx_versus_tree_tree_id ON public.versus_tree(tree_id);
